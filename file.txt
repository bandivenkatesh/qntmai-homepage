        function animateSlogan() {
            gsap.to(sloganText, {
                opacity: 0,
                duration: 0.5,
                onComplete: () => {
                    currentSloganIndex = (currentSloganIndex + 1) % slogans.length;
                    sloganText.textContent = slogans[currentSloganIndex];
                    gsap.to(sloganText, {
                        opacity: 1,
                        duration: 0.8,
                        delay: 0.2, // Small delay for flickering effect
                        repeat: 1, // Flicker in and out once

                            gsap.to(sloganText, {
                                opacity: 1,
                                duration: 0.5,
                                delay: 2.5, // Display time before next flicker, slightly longer
                                onComplete: animateSlogan
                            });
                        }
                    });
                }
            });
        }

        function initThreeJS() {

            scene = new THREE.Scene();

            // --- Central 3D Object (Icosahedron/Digital Brain) ---
            const geometry = new THREE.IcosahedronGeometry(1.5, 1); // Size and detail

            scene.add(sphere);

            // Add a wireframe to the sphere for a more digital look
            const wireframeGeometry = new THREE.WireframeGeometry(geometry);
            const wireframeMaterial = new THREE.LineBasicMaterial({

                linewidth: 3, // Thicker lines
                transparent: true,
                opacity: 0.7 // More visible
            });
            const wireframe = new THREE.LineSegments(wireframeGeometry, wireframeMaterial);

            // --- Background Particle System (Starfield/Space Dust) ---
            const particleCount = 1500; // More particles
            const particlesGeometry = new THREE.BufferGeometry();
            const positions = [];

                colors.push(color.r, color.g, color.b);

            particlesGeometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));

            mouseTrailGeometry.setAttribute('opacity', new THREE.BufferAttribute(mouseTrailOpacities, 1));
            mouseTrailGeometry.setAttribute('color', new THREE.BufferAttribute(mouseTrailColors, 3));

            const mouseTrailMaterial = new THREE.PointsMaterial({

                    `void main() {

                    `
                );
                shader.fragmentShader = `
                    varying float vOpacity;
                    ${shader.fragmentShader}
                `.replace(
                    `gl_FragColor = vec4( outgoingLight, diffuseColor.a );`,
                    `gl_FragColor = vec4( outgoingLight, diffuseColor.a * vOpacity );`
                );
            };

                blending: THREE.AdditiveBlending
            });
            dataGrid = new THREE.Mesh(gridGeometry, gridMaterial);
            dataGrid.rotation.x = -Math.PI / 2; // Lay flat
            dataGrid.position.y = -10; // Below the main content
            dataGrid.position.z = -50; // Further back
            scene.add(dataGrid);


            // --- Lighting ---
            const ambientLight = new THREE.AmbientLight(0x404040, 0.6); // Slightly brighter ambient
            scene.add(ambientLight);

            const pointLight = new THREE.PointLight(0x00ffff, 1.2, 100); // Cyber green point light, brighter
            pointLight.position.set(5, 5, 5);
            scene.add(pointLight);

            const pointLight2 = new THREE.PointLight(0xff00ff, 1.2, 100); // Magenta point light, brighter
            pointLight2.position.set(-5, -5, -5);
            scene.add(pointLight2);

            // Additional dynamic lights
            pointLight3 = new THREE.PointLight(0x8a2be2, 0.8, 80); // Violet light
            pointLight3.position.set(0, 0, 10);
            scene.add(pointLight3);

            pointLight4 = new THREE.PointLight(0x33ff33, 0.7, 70); // Bright green light
            pointLight4.position.set(0, 0, -10);
            scene.add(pointLight4);


            // --- Event Listeners ---
            document.addEventListener('mousemove', onDocumentMouseMove, false);
            window.addEventListener('resize', onWindowResize, false);
        }

        function onDocumentMouseMove(event) {
            mouseX = (event.clientX - windowHalfX) * 0.001;
            mouseY = (event.clientY - windowHalfY) * 0.001;

            // Update mouse trail particles
            const vector = new THREE.Vector3(
                (event.clientX / window.innerWidth) * 2 - 1,
                -(event.clientY / window.innerHeight) * 2 + 1,
                0.5
            );
            vector.unproject(camera);
            const dir = vector.sub(camera.position).normalize();
            const distance = -camera.position.z / dir.z;
            const pos = camera.position.clone().add(dir.multiplyScalar(distance));

            // Shift existing particles and add new one
            const positions = mouseTrailParticles.geometry.attributes.position.array;
            const opacities = mouseTrailParticles.geometry.attributes.opacity.array;

            for (let i = 0; i < mouseTrailCount - 1; i++) {
                positions[i * 3] = positions[(i + 1) * 3];
                positions[i * 3 + 1] = positions[(i + 1) * 3 + 1];
                positions[i * 3 + 2] = positions[(i + 1) * 3 + 2];
                opacities[i] = opacities[i + 1] * 0.95; // Fade out
            }

            positions[(mouseTrailCount - 1) * 3] = pos.x;
            positions[(mouseTrailCount - 1) * 3 + 1] = pos.y;
            positions[(mouseTrailCount - 1) * 3 + 2] = pos.z;
            opacities[mouseTrailCount - 1] = 1.0; // New particle is fully opaque

            mouseTrailParticles.geometry.attributes.position.needsUpdate = true;
            mouseTrailParticles.geometry.attributes.opacity.needsUpdate = true;
        }

        function onWindowResize() {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        }

        // --- Animation Loop ---
        function animate() {
            requestAnimationFrame(animate);

            // Rotate the central sphere
            sphere.rotation.x += 0.002;
            sphere.rotation.y += 0.005;

            // Subtle reaction of sphere to mouse
            const targetEmissiveIntensity = 0.7 + (Math.abs(mouseX) + Math.abs(mouseY)) * 0.5;
            sphere.material.emissiveIntensity = gsap.utils.interpolate(sphere.material.emissiveIntensity, targetEmissiveIntensity, 0.05);

            // Move background particles (simulate space dust/drift)
            particles.rotation.x += 0.0001;
            particles.rotation.y += 0.0002;
            particles.position.z += 0.02; // Move particles faster
            if (particles.position.z > 50) {
                particles.position.z = -50;
            }

            // Animate data grid
            dataGrid.rotation.z += 0.001; // Rotate grid slightly
            dataGrid.position.z += 0.1; // Move grid forward
            if (dataGrid.position.z > 50) {
                dataGrid.position.z = -50;
            }
            // Make data grid opacity react to scroll
            const scrollYNormalizedGrid = window.scrollY / (document.body.scrollHeight - window.innerHeight);
            dataGrid.material.opacity = 0.08 + (scrollYNormalizedGrid * 0.05); // Increase opacity slightly on scroll
            dataGrid.material.opacity = Math.min(0.15, dataGrid.material.opacity); // Clamp max opacity


            // Animate additional lights
            const time = Date.now() * 0.0005;
            pointLight3.position.x = Math.sin(time * 0.7) * 15;
            pointLight3.position.y = Math.cos(time * 0.5) * 15;
            pointLight3.position.z = Math.sin(time * 0.3) * 15;

            pointLight4.position.x = Math.cos(time * 0.9) * 12;
            pointLight4.position.y = Math.sin(time * 0.6) * 12;
            pointLight4.position.z = Math.cos(time * 0.4) * 12;

            // Camera movement based on mouse position (parallax effect)
            camera.position.x += (mouseX * 7 - camera.position.x) * 0.05; // Stronger parallax
            camera.position.y += (-mouseY * 7 - camera.position.y) * 0.05;
            camera.lookAt(scene.position);

            // Camera zoom based on scroll
            const scrollYNormalized = window.scrollY / (document.body.scrollHeight - window.innerHeight);
            camera.position.z = 5 - (scrollYNormalized * 3); // Zoom out slightly as user scrolls down
            camera.position.z = Math.max(2, Math.min(5, camera.position.z)); // Clamp values


            renderer.render(scene, camera);
        }

        // --- Magnetic Button Effect ---
        const launchButton = document.getElementById('launch-portfolio-button');

        function applyMagneticEffect(button) {
            const buttonRect = button.getBoundingClientRect();
            let buttonCenterX = buttonRect.left + buttonRect.width / 2;
            let buttonCenterY = buttonRect.top + buttonRect.height / 2;

            // Recalculate center on scroll/resize
            window.addEventListener('resize', () => {
                const newRect = button.getBoundingClientRect();
                buttonCenterX = newRect.left + newRect.width / 2;
                buttonCenterY = newRect.top + newRect.height / 2;
            });

            document.addEventListener('mousemove', (e) => {
                const mouseX = e.clientX;
                const mouseY = e.clientY;

                const distanceX = mouseX - buttonCenterX;
                const distanceY = mouseY - buttonCenterY;
                const distance = Math.sqrt(distanceX * distanceX + distanceY * distanceY);

                const maxDistance = 180; // Increased max distance for magnetic effect
                const strength = 0.4; // Stronger pull

                if (distance < maxDistance) {
                    const normalizedDistance = distance / maxDistance;
                    const inverseNormalizedDistance = 1 - normalizedDistance; // Closer = stronger effect

                    const translateX = (distanceX / distance) * (inverseNormalizedDistance * strength * 30); // More movement
                    const translateY = (distanceY / distance) * (inverseNormalizedDistance * strength * 30);

                    gsap.to(button, {
                        x: translateX,
                        y: translateY,
                        ease: "power3.out", // Smoother ease
                        duration: 0.3
                    });
                } else {
                    gsap.to(button, {
                        x: 0,
                        y: 0,
                        ease: "elastic.out(1, 0.6)", // Slightly bouncier
                        duration: 0.8
                    });
                }
            });
        }

        applyMagneticEffect(launchButton);
